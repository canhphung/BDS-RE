cmake_minimum_required(VERSION 3.20)
project(BDS_RE VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)

option(BDS_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(Threads REQUIRED)
# Vendored zlib 1.3.1 (zlib license), deflate/inflate sources only; see third_party/README.md.
add_library(bds_zlib STATIC third_party/zlib/adler32.c third_party/zlib/crc32.c third_party/zlib/deflate.c
 third_party/zlib/inffast.c third_party/zlib/inflate.c third_party/zlib/inftrees.c third_party/zlib/trees.c third_party/zlib/zutil.c)
target_include_directories(bds_zlib SYSTEM PUBLIC third_party/zlib)
if(MSVC)
 target_compile_definitions(bds_zlib PRIVATE _CRT_SECURE_NO_DEPRECATE _CRT_NONSTDC_NO_DEPRECATE)
endif()
set_target_properties(bds_zlib PROPERTIES POSITION_INDEPENDENT_CODE ON)
# Vendored Mbed TLS 3.6.2 (Apache-2.0) subset: ECDSA/ECDH P-384, SHA-2, AES, CTR-DRBG.
set(BDS_MBEDTLS_SOURCES aes asn1parse asn1write bignum bignum_core constant_time ctr_drbg ecdh ecdsa ecp ecp_curves
 entropy entropy_poll md platform_util sha256 sha512)
list(TRANSFORM BDS_MBEDTLS_SOURCES PREPEND third_party/mbedtls/library/)
list(TRANSFORM BDS_MBEDTLS_SOURCES APPEND .c)
add_library(bds_mbedtls STATIC ${BDS_MBEDTLS_SOURCES})
target_include_directories(bds_mbedtls SYSTEM PUBLIC third_party/mbedtls/include third_party/mbedtls_config
 PRIVATE third_party/mbedtls/library)
target_compile_definitions(bds_mbedtls PUBLIC MBEDTLS_CONFIG_FILE="bds_mbedtls_config.h")
set_target_properties(bds_mbedtls PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(WIN32)
 target_link_libraries(bds_mbedtls PRIVATE bcrypt)
endif()
add_library(bds_core src/core/runtime.cpp src/core/config.cpp src/core/atomic_file.cpp src/world/store.cpp src/world/water.cpp src/world/terrain.cpp src/world/noise.cpp src/world/biome_layers.cpp src/world/biome_data.cpp src/world/overworld.cpp src/world/decoration.cpp src/world/seed.cpp src/world/vegetation.cpp

target_include_directories(bds_core PUBLIC include)
target_include_directories(bds_core SYSTEM PRIVATE third_party)
target_link_libraries(bds_core PUBLIC Threads::Threads PRIVATE bds_zlib bds_mbedtls)
if(WIN32)
 target_link_libraries(bds_core PRIVATE ws2_32)
endif()
if(MSVC)
 target_compile_options(bds_core PRIVATE /W4 /permissive-)
else()
 target_compile_options(bds_core PRIVATE -Wall -Wextra -Wpedantic)
 if(BDS_SANITIZE)
  target_compile_options(bds_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(bds_core PUBLIC -fsanitize=address,undefined)
 endif()
endif()
