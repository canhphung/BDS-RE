cmake_minimum_required(VERSION 3.20)
project(BDS_RE VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BDS_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(Threads REQUIRED)
add_library(bds_core src/core/runtime.cpp src/core/config.cpp)
target_include_directories(bds_core PUBLIC include)
target_link_libraries(bds_core PUBLIC Threads::Threads)
if(MSVC)
 target_compile_options(bds_core PRIVATE /W4 /permissive-)
else()
 target_compile_options(bds_core PRIVATE -Wall -Wextra -Wpedantic)
 if(BDS_SANITIZE)
  target_compile_options(bds_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(bds_core PUBLIC -fsanitize=address,undefined)
 endif()
endif()
add_executable(bds_server src/main.cpp)
target_link_libraries(bds_server PRIVATE bds_core)
include(CTest)
if(BUILD_TESTING)
 add_executable(core_tests tests/unit/core_tests.cpp)
 target_link_libraries(core_tests PRIVATE bds_core)
 add_test(NAME core COMMAND core_tests)
 add_executable(world_tests tests/unit/world_tests.cpp)
 target_link_libraries(world_tests PRIVATE bds_core)
 add_test(NAME world COMMAND world_tests)
 add_test(NAME lifecycle COMMAND bds_server --ticks 3 --tick-ms 1)
 set_tests_properties(lifecycle PROPERTIES PASS_REGULAR_EXPRESSION "stopped ticks=3" TIMEOUT 10)
 add_test(NAME invalid_config COMMAND bds_server --tick-ms 0)
 set_tests_properties(invalid_config PROPERTIES WILL_FAIL TRUE TIMEOUT 10)
endif()
