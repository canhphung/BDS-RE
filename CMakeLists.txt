cmake_minimum_required(VERSION 3.20)
project(BDS_RE VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BDS_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(Threads REQUIRED)
# Vendored zlib 1.3.1 (zlib license), deflate/inflate sources only; see third_party/README.md.
add_library(bds_zlib STATIC third_party/zlib/adler32.c third_party/zlib/crc32.c third_party/zlib/deflate.c
 third_party/zlib/inffast.c third_party/zlib/inflate.c third_party/zlib/inftrees.c third_party/zlib/trees.c third_party/zlib/zutil.c)
target_include_directories(bds_zlib SYSTEM PUBLIC third_party/zlib)
if(MSVC)
 target_compile_definitions(bds_zlib PRIVATE _CRT_SECURE_NO_DEPRECATE _CRT_NONSTDC_NO_DEPRECATE)
endif()
set_target_properties(bds_zlib PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(bds_core src/core/runtime.cpp src/core/config.cpp src/core/atomic_file.cpp src/world/store.cpp src/core/console.cpp
 src/server/properties.cpp src/server/access.cpp src/server/backup.cpp src/server/control.cpp src/network/discovery.cpp src/network/login.cpp src/network/raknet.cpp
 src/network/binary.cpp src/network/batch.cpp src/network/connection.cpp)
target_include_directories(bds_core PUBLIC include)
target_include_directories(bds_core SYSTEM PRIVATE third_party)
target_link_libraries(bds_core PUBLIC Threads::Threads PRIVATE bds_zlib)
if(WIN32)
 target_link_libraries(bds_core PRIVATE ws2_32)
endif()
if(MSVC)
 target_compile_options(bds_core PRIVATE /W4 /permissive-)
else()
 target_compile_options(bds_core PRIVATE -Wall -Wextra -Wpedantic)
 if(BDS_SANITIZE)
  target_compile_options(bds_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(bds_core PUBLIC -fsanitize=address,undefined)
 endif()
endif()
add_executable(bds_server src/main.cpp)
target_link_libraries(bds_server PRIVATE bds_core)
include(CTest)
if(BUILD_TESTING)
 add_executable(core_tests tests/unit/core_tests.cpp)
 target_link_libraries(core_tests PRIVATE bds_core)
 add_test(NAME core COMMAND core_tests)
 add_executable(server_tests tests/unit/server_tests.cpp)
 target_link_libraries(server_tests PRIVATE bds_core)
 add_test(NAME server COMMAND server_tests ${CMAKE_CURRENT_SOURCE_DIR}/server.properties)
 add_executable(discovery_tests tests/protocol_discovery.cpp)
 target_link_libraries(discovery_tests PRIVATE bds_core)
 add_test(NAME discovery COMMAND discovery_tests)
 add_executable(raknet_tests tests/protocol_raknet.cpp)
 target_link_libraries(raknet_tests PRIVATE bds_core)
 add_test(NAME raknet COMMAND raknet_tests)
 add_executable(login_tests tests/protocol_login.cpp)
 target_link_libraries(login_tests PRIVATE bds_core)
 add_test(NAME login_codec COMMAND login_tests)
 add_executable(batch_tests tests/protocol/batch_tests.cpp)
 target_link_libraries(batch_tests PRIVATE bds_core)
 target_include_directories(batch_tests SYSTEM PRIVATE third_party)
 add_test(NAME batch COMMAND batch_tests ${CMAKE_CURRENT_SOURCE_DIR}/reverse/evidence/packet_ids.json)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME raknet_handshake COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/raknet_handshake.py $<TARGET_FILE:bds_server>)
 set_tests_properties(raknet_handshake PROPERTIES TIMEOUT 45)
 add_test(NAME server_console COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/server_console.py $<TARGET_FILE:bds_server>)
 set_tests_properties(server_console PROPERTIES TIMEOUT 45)
 add_test(NAME runtime_failure COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/runtime_failure.py $<TARGET_FILE:bds_server>)
 set_tests_properties(runtime_failure PROPERTIES TIMEOUT 20)
 add_executable(world_tests tests/unit/world_tests.cpp)
 target_link_libraries(world_tests PRIVATE bds_core)
 add_test(NAME world COMMAND world_tests)
 add_executable(chunk_tests tests/unit/chunk_tests.cpp)
 target_link_libraries(chunk_tests PRIVATE bds_core)
 add_test(NAME chunks COMMAND chunk_tests)
 add_executable(persistence_tests tests/unit/persistence_tests.cpp)
 target_link_libraries(persistence_tests PRIVATE bds_core)
 add_test(NAME persistence COMMAND persistence_tests)
 add_test(NAME world_lifecycle COMMAND ${CMAKE_COMMAND}
  -DSERVER=$<TARGET_FILE:bds_server> -DTEST_ROOT=${CMAKE_CURRENT_BINARY_DIR}
  -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/world_lifecycle.cmake)
 set_tests_properties(world_lifecycle PROPERTIES TIMEOUT 15)
 add_test(NAME lifecycle COMMAND bds_server --ticks 3 --tick-ms 1 --network 0)
 set_tests_properties(lifecycle PROPERTIES PASS_REGULAR_EXPRESSION "stopped ticks=3" TIMEOUT 10)
 add_test(NAME invalid_config COMMAND bds_server --tick-ms 0)
 set_tests_properties(invalid_config PROPERTIES WILL_FAIL TRUE TIMEOUT 10)
endif()
